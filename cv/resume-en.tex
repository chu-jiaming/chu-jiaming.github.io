\sectionTitle{Research}{\faFlask}
%======================================================================
\begin{experiences}
  \experience%
    {2026}%
    {\projecttitle{VGGT-Splat: Can VGGT Replace COLMAP for 3DGS Reconstruction} \hfill
     \pagelink{https://chu-jiaming.github.io/vggt-splat/}{page}\enspace
     \codelink{https://github.com/chu-jiaming/VGGT-Splat}{code}}%
    [\begin{itemize}
      \item Problem: 3DGS from phone video first needs camera poses. How many frames do COLMAP and feed-forward VGGT need? Does a VGGT depth prior help with few frames?
      \item Method (independent project): 6 scenes $\times$ 8/16/32/all frames $\times$ 3 methods, 72 runs.
      \item Result 1: \keypoint{VGGT is more reliable with few frames; COLMAP is better with all frames.} COLMAP fails on 4 of 6 scenes at 8 frames and 3 of 6 at 16, VGGT never;
        with all frames COLMAP leads by 2.1--4.6\,dB on five scenes, likely because VGGT's poses are less precise.
      \item Result 2: \keypoint{The depth prior barely improves novel views.} With few frames the error sits in regions no training view sees, which a depth loss cannot reach.
    \end{itemize}]
  \separator{0.5ex}
  \experience%
    {2026}%
    {\projecttitle{PoseGuard-NeRF: When Does Joint Pose Optimization Help NeRF} \hfill
     \pagelink{https://chu-jiaming.github.io/poseguard-nerf/}{page}\enspace
     \codelink{https://github.com/chu-jiaming/PoseGuard-NeRF}{code}}%
    [\begin{itemize}
      \item Problem: NeRF needs accurate camera poses. Joint pose optimization can correct errors, but when does it help and when does it hurt?
      \item Method (independent project): Lego scene, 9 pose conditions $\times$ 5 methods, 45 vanilla-NeRF runs; Sim(3) alignment separates worse reconstruction from coordinate drift.
      \item Result 1: \keypoint{With noisy poses, joint optimization clearly helps.} At 5\% translation noise, PSNR rises from 14.1\,dB (fixed cameras) to 22.7\,dB (with warm-up).
      \item Result 2: \keypoint{With accurate poses, it lowers PSNR, mostly through global drift.} Without warm-up PSNR drops 27.1 $\rightarrow$ 17.9\,dB; Sim(3) alignment recovers 5.5\,dB.
    \end{itemize}]
  \separator{0.5ex}
  \experience%
    {2026}%
    {\projecttitle{AFVE-Bench: Abandoned-Frame Verbalization} \privatenote{noteen}\hfill
     \paperlink{https://chu-jiaming.github.io/papers/afve-bench.pdf}{preprint}}%
    [\begin{itemize}
      \item Problem: asked to rewrite under a new frame, an LLM may adopt it yet still expose the abandoned one. Existing benchmarks mainly check the latest target and can miss this.
      \item Method (first author): 192 bilingual four-turn dialogues across 8 writing domains; 4,204 responses from 11 model configurations, blind double-annotated and adjudicated.
      \item Result: \keypoint{Meeting the new target does not stop the old frame from showing.} 25.5\% of responses after old-frame turns show it; so do 19.8\% of those meeting the target.
    \end{itemize}]
\end{experiences}
